# A grammar file, coloured by nexus.grammar's @highlight.
@lexer

tokens
    ident, number, string, eof, err

'"' ([^"\\\n] | '\\' .)* '"'    → string
[0-9]+                          → number
'\n'                            → skip, skip
[a-z]+ / '('                    → ident, rewind(1)
.                               → err

@parser

@display
    IDENT: "a name\tthere", "=": "'='"

list! = item+                   → (list ...1)
item  = name:IDENT "=" NUMBER   → (set 1 3)
      | STRING? -> (str ~1)
