list    = L(item)                   → (list ...1)
        | "(" L(item, ",") ")"      → (sep ...2)
        | L(item?)                  → (optitems ...1)
        | L(item?, ";")             → (optsep ...1)
        | L(item, SEMI)             → (toksep ...1)
opts    = A [B C] D                 → (opts 1 2 3 4)
        | [L(item)]                 → (optlist ...1)
        | [item ...]                  → (many ...1)
        | [B | C]                   → (alts 1)
groups  = (A | B C) D               → (group 1 2)
        | (cmd !SPACES?)+           → (cmds ...1)
quants  = A? B* C+                  → (q 1 2 3)
        | !A B                      → (skip 2)
        | !A* B                     → (skipq 2)
hints   = glvn X "#" X ":" expr     → (hint 1 2)
        | A B <                     → (tight 1 2)
        | A B >                     → (loose 1 2)
refs    = @infix IDENT "lit" TOKEN  → (refs 1 2 3 4)
